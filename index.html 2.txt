    </div>

    <div id="galleryScreen" class="gallery-container">
        <button id="backBtn" class="back-btn" style="display: none;" onclick="showOptionsAgain()">← Gruplara Geri Dön</button>

<footer>
    <input type="file" id="fileInput" multiple accept="image/*" onchange="handleFiles(this.files)">
    <button id="sortBtn" class="btn" style="display: none;" onclick="sortPhotos()">Fotoğrafları sıraya diz</button>
</footer>

<script>
    let allPhotos = [];
    let activePhotos = [];

    function requestPermission() {
        document.getElementById('fileInput').click();
    }

    function handleFiles(files) {
        if (files.length > 0) {
            allPhotos = [];
            for (let i = 0; i < files.length; i++) {
                allPhotos.push({
                    file: files[i],
                    url: URL.createObjectURL(files[i]),
                    name: files[i].name,
                    lastModified: files[i].lastModified
                });
            }

            document.getElementById('permissionScreen').style.display = 'none';

            if (allPhotos.length <= 10) {
                activePhotos = allPhotos;
                document.getElementById('galleryScreen').style.display = 'block';
                document.getElementById('sortBtn').style.display = 'inline-block';
                renderPhotos();
            } else {
                setupOptions();
            }
        }
    }

    function setupOptions() {
        const optionsList = document.getElementById('optionsList');
        optionsList.innerHTML = '';
        
        const mid = Math.floor(allPhotos.length / 2);
        const group1 = allPhotos.slice(0, mid);
        const group2 = allPhotos.slice(mid);

        const btn1 = document.createElement('button');
        btn1.className = 'option-btn';
        btn1.innerText = `Fotoğraf Grubu 1 (${group1.length} Fotoğraf)`;
        btn1.onclick = () => showGroup(group1);

        const btn2 = document.createElement('button');
        btn2.className = 'option-btn';
        btn2.innerText = `Fotoğraf Grubu 2 (${group2.length} Fotoğraf)`;
        btn2.onclick = () => showGroup(group2);

        optionsList.appendChild(btn1);
        optionsList.appendChild(btn2);

        document.getElementById('optionsScreen').style.display = 'block';
    }

    function showGroup(group) {
        activePhotos = group;
        document.getElementById('optionsScreen').style.display = 'none';
        document.getElementById('galleryScreen').style.display = 'block';
        document.getElementById('backBtn').style.display = 'inline-block';
        document.getElementById('sortBtn').style.display = 'inline-block';
        renderPhotos();
    }

    function showOptionsAgain() {
        document.getElementById('galleryScreen').style.display = 'none';
        document.getElementById('sortBtn').style.display = 'none';
        document.getElementById('optionsScreen').style.display = 'block';
    }

    function renderPhotos() {
        const grid = document.getElementById('photoGrid');
        grid.innerHTML = '';
        
        activePhotos.forEach(photo => {
            const card = document.createElement('div');
            card.className = 'photo-card';
            
            const dateObj = new Date(photo.lastModified);
            const dateStr = dateObj.toLocaleDateString() + ' ' + dateObj.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

            card.innerHTML = `
                <img src="${photo.url}" alt="${photo.name}">
                <div class="photo-info" title="${photo.name}">${dateStr}</div>
            `;
            grid.appendChild(card);
        });
    }

    function sortPhotos() {
        activePhotos.sort((a, b) => a.lastModified - b.lastModified);
        renderPhotos();
    }
</script>
